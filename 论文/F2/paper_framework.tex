\begin{abstract}
    Minimizing $w_j C_{\max}$ on a two-machine flow shop,
    denoted $F2 \parallel w_j C_{\max}$, is a classical NP-hard scheduling problem.
    Existing exact methods rely on enumerative techniques that scale poorly with instance size,
    while approximation algorithms often impose restrictive assumptions on processing times.
    We first sequence all jobs globally by Johnson's rule, obtaining an initial permutation $S$.
    Scanning $S$ from left to right, we partition it into at most $K$ blocks: whenever a weight not seen before appears, it serves as the final job of the current block, and the next job starts a new block.
    We establish structural properties of optimal schedules: the permutation property, the positional constraint that block $\mathcal{G}_k$ jobs cannot appear beyond the first $k$ blocks,
    and the block-end condition requiring the final job of each block to carry that block's weight (automatically satisfied by construction).
    The intra-block Johnson ordering inherited from $S$ is further formalized within the DP framework.
    Building on these properties and the interval-based dynamic programming framework \cite{hall1998} developed for $F2|r_j|C_{\max}$,
    we design a polynomial-time dynamic programming algorithm with $O(|Y|_{\max}\cdot n)$ complexity that appends jobs block by block.
    We further show that geometric rounding of weights and processing times converts this algorithm into a polynomial-time approximation scheme.
\end{abstract}

\noindent \textbf{Keywords:} Flow shop scheduling; $w_j C_{\max}$; Dynamic programming; Structural properties

\section{Preliminaries}
\label{sec:preliminaries}


We consider the problem $F2 \parallel w_j C_{\max}$.
Let $\mathcal{J} = \{J_1, J_2, \dots, J_n\}$ be the set of $n$ jobs to be processed on two machines M1 and M2 in series:
 each job $J_j$ must be processed first on M1 for $a_j$ time units and then on M2 for $b_j$ time units, with no preemption allowed.
M1 and M2 can process at most one job at a time.
Each job $J_j$ has a weight $w_j > 0$.
A permutation schedule $S = (\pi(1), \pi(2), \dots, \pi(n))$ specifies the processing order on both machines.
For a permutation $S$, let $A_i = \sum_{h=1}^{i} a_{\pi(h)}$ be the cumulative M1 completion time after the first $i$ jobs.
The makespan $C_{\max}(S)$ is the M2 completion time of the last job, defined recursively by
\begin{equation}\label{eq:makespan}
    \qquad
    C_i = \max\{\, C_{i-1},\; A_i \,\} + b_i,\;\; i = 1,\dots,n,
\end{equation}
and $C_0 = 0$, $C_{\max}(S) = C_n$.
The objective is to find a permutation $S$ that minimizes $w_j C_{\max}$.

\begin{theorem}\label{thm:nphard}
    $F2 \parallel w_j C_{\max}$ is strongly NP-hard.
\end{theorem}
\noindent \textbf{Proof.}  \hfill$\Box$\\


In a permutation schedule, the job order on M1 and M2 is identical for $F2 \parallel w_j C_{\max}$ there always exists an optimal schedule of this form \cite{baker1974introduction}.
It is known from  (\citealp{johnson1954optimal}) that when the number of machines is less than or equal to 3, we have all $n$ jobs are sequenced by Johnson's rule to obtain a global sequence $S = (\pi(1), \pi(2), \dots, \pi(n))$.
Scanning $S$ from left to right, we partition it into $K$ blocks by the first occurrence of each new weight: whenever a weight not seen before appears, it serves as the final job of the current block, and the next job starts a new block.
Let $W(\mathcal{G}_k)$ denote the weight carried by the final job of block $\mathcal{G}_k$.

We first establish a positional constraint: in any optimal schedule, for each $k = 1, \dots, K$, the jobs belonging to block $\mathcal{G}_k$
can only appear within the first $k$ blocks of the final schedule.
Consequently, once the $k$-th block of the final schedule has been processed, all jobs from the first $k$ blocks of $S$ are fully scheduled,
and the $k$-th final block may contain only jobs from $\mathcal{G}_1, \mathcal{G}_2, \dots, \mathcal{G}_K$.
We further establish a block-end condition: in any optimal schedule, the final job of the $k$-th final block must carry weight $W(\mathcal{G}_k)$.
By construction, $W(\mathcal{G}_k)$ is precisely the weight that first appears at the end of $\mathcal{G}_k$ in $S$, so this condition is automatically satisfied.

\begin{exam}\label{ex:blocking}
Consider a 6-job instance with Johnson sequence $S = (J_1, \dots, J_6)$ and weights
$(w_1,\dots,w_6) = (3,\,1,\,1,\,3,\,4,\,2)$.
Scanning $S$ from left to right and cutting whenever a weight not seen before appears yields the following partition:
\[
\underbrace{J_1,\; J_2}_{\mathcal{G}_1}\;
\underbrace{J_3,\; J_4,\; J_5}_{\mathcal{G}_2}\;
\underbrace{J_6}_{\mathcal{G}_3}
\qquad
\]
The resulting blocks are $\mathcal{G}_1=\{J_1,J_2\}$, $\mathcal{G}_2=\{J_3,J_4,J_5\}$, $\mathcal{G}_3=\{J_6\}$, with block weights $W(\mathcal{G}_1)=1$, $W(\mathcal{G}_2)=4$, $W(\mathcal{G}_3)=2$, where each $W(\mathcal{G}_k)$ is the new weight that triggered the cut.
\end{exam}




\section{Dynamic Programming Algorithm}
\label{sec:dp_algorithm}

The structural properties established in Section~\ref{sec:preliminaries} enable a dynamic programming algorithm that constructs the schedule by appending jobs block by block.
Our approach adapts the interval-based state representation of \cite{hall1998}, originally devised for the $F2|r_j|C_{\max}$, to the $w_j C_{\max}$ objective.

As described in Section~\ref{sec:preliminaries}, the algorithm takes as input the global Johnson sequence $S = (\pi(1), \pi(2), \dots, \pi(n))$ and its partition into $K$ blocks $\mathcal{G}_1, \dots, \mathcal{G}_K$, and processes jobs in block order $\mathcal{G}_1, \mathcal{G}_2, \dots, \mathcal{G}_K$, appending each job to the end of the current partial permutation.
The positional constraint is automatically satisfied by this discipline: jobs of $\mathcal{G}_k$ are appended only after blocks $\mathcal{G}_1, \dots, \mathcal{G}_{k-1}$ have been fully processed.
The block-end condition is enforced at block boundaries via the check $W(\mathcal{G}_k) \le L$ described below.
Its correctness follows from the fact that $W(\mathcal{G}_k)$ is the weight of the final job of block $\mathcal{G}_k$ and $L$ records the weight of the job that achieves the maximum $w_j C_j$ so far; the inequality ensures block weights are non-increasing: a block with larger weight appearing later would inflate the weighted makespan $w_j C_{\max}$ beyond the optimum.
If violated, no subsequent appending from $\mathcal{G}_{k+1}, \dots, \mathcal{G}_K$ can alter the already-fixed weight ordering of the prefix, so the state can be safely discarded.
If the check passes and the current job refreshes the maximum weighted completion time, $L$ is updated to $w_j$.

\begin{lemma}\label{lem:spt_lpt}
Within each block $\mathcal{G}_k$, all jobs are sequenced according to Johnson's rule \cite{johnson1954optimal}:
for any two jobs $x, y \in \mathcal{G}_k$, job $x$ precedes job $y$ if and only if
\[
    \min\{a_x, b_y\} \le \min\{a_y, b_x\},
\]
with ties broken by non-decreasing $a_j$.
Since the initial sequence $S$ already satisfies this ordering globally and each $\mathcal{G}_k$ is a contiguous subsequence of $S$,
the intra-block Johnson order is automatically preserved by the append-in-block-order discipline.
\end{lemma}

\subsection{State Definition and Dynamic Programming}

Each state implicitly corresponds to a partial permutation $S$ uniquely determined by the DP construction path; $S$ is not stored.
A state is a 4-tuple $(A,\, B,\, C,\, L)$, where $A$ and $B$ are the cumulative completion times on M1 and M2, respectively, $C = B$ is the makespan of the partial schedule, and $L$ records the weight of the job that achieves the maximum $w_j C_j$ among all jobs in the current partial schedule.
The initial state is $(0,\, 0,\, 0,\, 0)$, corresponding to $S = \emptyset$; $L = 0$ means no job has been scheduled yet.

Let $\mathcal{Y}$ denote the set of states after processing a given prefix of jobs.
Given a state $(A,\, B,\, C,\, L) \in \mathcal{Y}$ and a job $j \in \mathcal{G}_k$ to append, the transition produces a new state $(A',\, B',\, C',\, L')$ defined by
\begin{align}
    A' &= A + a_j, \label{eq:trans-A}\\
    B' &= \max(B,\, A') + b_j, \label{eq:trans-B}\\
    C' &= B', \label{eq:trans-C}\\
    L' &= \begin{cases}
        w_j, & \text{if $w_j \cdot C' > L \cdot C$},\\
        L,  & \text{otherwise.}
    \end{cases} \label{eq:trans-L}
\end{align}
All four operations take $O(1)$ time.
At a block boundary (when the last job of $\mathcal{G}_k$ is appended), the constraint $W(\mathcal{G}_k) \le L'$ is verified; if violated, the candidate state is discarded.

\begin{lemma}\label{lem:dominance}
For any two states $(A, B, C, L)$ and $(A', B', C', L')$ in $\mathcal{Y}$ at the same processing stage satisfying $C \le C'$ and $L \le L'$, the latter state can be removed from $\mathcal{Y}$.
\end{lemma}
\noindent\textbf{Proof.}
Let $S$ and $S'$ be the partial schedules attaining the two states.
Let $\sigma$ be an arbitrary suffix schedule for the remaining jobs.
The contribution of $\sigma$ to the makespan is monotone non-decreasing in the prefix completion times $(A, B)$; the condition $C \le C'$ guarantees that $S$ leads to a makespan no larger than $S'$ under any suffix.
Moreover, $L \le L'$ ensures that the maximum weighted completion time $w_j C_j$ among already scheduled jobs is no larger in $S$ than in $S'$.
Therefore $S$ dominates $S'$ under the $w_j C_{\max}$ objective.
\hfill$\Box$

The dominance rule of Lemma~\ref{lem:dominance} justifies retaining only Pareto-optimal states with respect to $(C, L)$ after processing each job: any state dominated by another in both $C$ and $L$ can be safely discarded.
The complete DP procedure is presented as Algorithm~1.

\smallskip
\noindent\rule{\textwidth}{0.4pt}
\vskip -2pt
{\noindent\textbf{Algorithm~1}~~DP\_F2\_wCmax}
\vskip -2pt
\noindent\rule{\textwidth}{0.4pt}
\vskip 1pt
\begin{algorithmic}[1]
    \State \textbf{Input:} Jobs $\mathcal{J}$ with processing times $(a_j, b_j)$ and weights $w_j$, $j = 1, \dots, n$; 
    Johnson sequence $S$ of $\mathcal{J} = \{J_1, J_2, \dots, J_n\}$; 
    partition parameters $\textit{start} = 1$, $K = 0$.
    \State \textbf{Output:} The optimal objective value $C^* = \max_j w_j C_j$ and its associated optimal permutation $S^*$.
    \For{$i = 2$ to $n$}
        \If{ $w_{S[i]}$ appears for the first time \textbf}
            \State $K \gets K + 1$; \quad $\mathcal{G}_K \gets S[\textit{start} \dots i]$
            \State $\textit{start} \gets i + 1$
        \EndIf
    \EndFor
    \If{$\textit{start} \le n$}
        \State $K \gets K + 1$; \quad $\mathcal{G}_K \gets S[\textit{start} \dots n]$
    \EndIf

    \State $\mathcal{Y} \gets \{ (0,\, 0,\, 0,\, 0) \}$

    \For{$k = 1$ to $K$}
        \State $w_k \gets W(\mathcal{G}_k)$
        \For{$i = 1$ to $|\mathcal{G}_k|$}
            \State $j \gets \mathcal{G}_k[i]$; \quad $\mathcal{Y}_{\text{new}} \gets \emptyset$
            \For{each $(A,\, B,\, C,\, L) \in \mathcal{Y}$}
                \State $A' \gets A + a_j$
                \State $B' \gets \max(B,\, A') + b_j$
                \State $C' \gets B'$
                \If{$w_j \cdot C' > L \cdot C$}
                    \State $L' \gets w_j$
                \Else
                    \State $L' \gets L$
                \EndIf
                \If{$i = |\mathcal{G}_k|$}
                    \If{$w_k > L'$} \textbf{continue} \EndIf
                \EndIf
                \State $\mathcal{Y}_{\text{new}} \gets \mathcal{Y}_{\text{new}} \cup \{ (A',\, B',\, C',\, L') \}$
            \EndFor
            \State Remove from $\mathcal{Y}_{\text{new}}$ any state $s$ dominated by some $s'$ with $s'.C \le s.C$ and $s'.L \le s.L$ (at least one strict)
            \State $\mathcal{Y} \gets \mathcal{Y}_{\text{new}}$
        \EndFor
    \EndFor

    \State $C^* \gets \min\{\, L \mid (A,\, B,\, C,\, L) \in \mathcal{Y} \,\}$
    \State Recover the optimal permutation $S^*$ by backtracking
    \State \Return $C^*$, $S^*$
\end{algorithmic}
