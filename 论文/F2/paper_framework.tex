\begin{abstract}
    Minimizing $w C_{\max}$ on a two-machine flow shop,
    denoted $F2 \parallel w C_{\max}$, is a classical NP-hard scheduling problem.
    Existing exact methods rely on enumerative techniques that scale poorly with instance size,
    while approximation algorithms often impose restrictive assumptions on processing times.
    We first sequence all jobs globally by Johnson's rule, obtaining an initial schedule $\pi$.
    Scanning $\pi$ from left to right, we partition it into at most $K$ blocks: whenever a weight not seen before appears, it serves as the final job of the current block, and the next job starts a new block.
    We establish structural properties of optimal schedules: the permutation property, the positional constraint that block $\mathcal{G}_k$ jobs cannot appear beyond the first $k$ blocks,
    and the block-end condition requiring the final job of each block to carry that block's weight (automatically satisfied by construction).
    The intra-block Johnson ordering inherited from $\pi$ is further formalized within the DP framework.
    Building on these properties and the interval-based dynamic programming framework (\citealp{hall1998}) developed for $F2|r_j|C_{\max}$,
    we design a polynomial-time dynamic programming algorithm with $O(|Y|_{\max}\cdot n)$ complexity that appends jobs block by block.
    We further show that geometric rounding of weights and processing times converts this algorithm into a polynomial-time approximation scheme.
\end{abstract}

\noindent \textbf{Keywords:} Flow shop scheduling; $w C_{\max}$; Dynamic programming; Structural properties

We consider the problem $F2 \parallel w C_{\max}$.
Let $\mathcal{J} = \{J_1, J_2, \dots, J_n\}$ be the set of $n$ jobs to be processed on two machines $M_1$ and $M_2$ in series:
 each job $J_j$ must be processed first on $M_1$ for $a_j$ time units and then on $M_2$ for $b_j$ time units, with no preemption allowed.
$M_1$ and $M_2$ can process at most one job at a time.
Each job $J_j$ has a weight $w_j > 0$.
A schedule $\pi = (\pi(1), \pi(2), \dots, \pi(n))$ specifies the processing order on both machines.
For a schedule $\pi$, let $A_j = \sum_{h=1}^{j} a_{\pi(h)}$ be the cumulative $M_1$ completion time after the first $j$ jobs.
The makespan $C_{\max}(\pi)$ is the $M_2$ completion time of the last job, defined recursively by
\begin{equation}\label{eq:makespan}
    \qquad
    C_j = \max\{\, C_{j-1},\; A_j \,\} + b_j,\;\; j = 1,\dots,n,
\end{equation}

and $C_0 = 0$, $C_{\max}(\pi) = C_n$.
The objective is to find a schedule $\pi$ that minimizes $w C_{\max}$.



For $F2 \parallel w C_{\max}$, a permutation schedule in which the job order on $M_1$ and $M_2$ ........ is optimal (\citealp{baker1974introduction}).
Johnson's rule (\citealp{johnson1954optimal}) sequences all $n$ jobs to minimize the makespan, .........

\begin{lemma}\label{lem:johnson}
For $F2 \parallel w C_{\max}$, there exists an optimal schedule in which all jobs are sequenced according to Johnson's rule.
\end{lemma}

\section{NP-hard}
\begin{theorem}\label{thm:nphard}
    $F2 \parallel w C_{\max}$ is NP-hard.
\end{theorem}
\noindent \textbf{Proof.}
We reduce from the Partition problem: given $n$ positive integers $x_1, x_2, \dots, x_r$ with $\sum_{i=1}^{r} x_i = 2T$, does there exist subset $Z_1$ and $Z_2 \subseteq \{1,\dots,r\}$ such that:
 $\sum_{i \in Z_1} x_i$ = $\sum_{i \in Z_2} x_i$ = $T$?

Construct an instance of $F2 \parallel w C_{\max}$,  Let the number of jobs $n = r + 1$.
Let $a_j = 1$, $b_j = rx_j$, $w_j = T+1$ for jobs $j = 1,\dots,n-1$.
Put $a_n = rT$, $b_n = 1$, $w_n = 2T+1$ for job $J_n$.
Does there exist a threshold Y such that:
\[
w C_{\max} \le Y = r(T+1)(2T+1)
\]
holds?

We partition jobs $j_1$ to $j_{n-1}$ into two subsets via $j_n$, 
the distribution of jobs on the machine is($\mathcal{J}_{j\epsilon A_1}, J_n, \mathcal{J}_{j\epsilon A_2}$). 
Since the processing times of $A_1$ and $A_2$ are unknown, while their total processing time is 2T. 
We may assume that the processing time of $A_1$ on $M_2$ is $r(T+\xi)$, $A_2$ is $r(T-\xi)$, where $-T\le\xi\le T$.
Therefore, the completion time of $J_n$ is determined by the following formula:
\begin{equation}\label{eq:completetime1}
    \qquad
    C_n^1 = |A_1| + rT + 1
\end{equation}
\begin{equation}\label{eq:completetime2}
    \qquad
    C_n^2 = 1 + r ( T + \xi ) + 1
\end{equation}
and $C_n = \max \{C_n^1, C_n^2\}$

If $\xi = 0$, then $\sum_{j\epsilon A_1}b_j =\sum_{j\epsilon A_2}b_j $, which means the partition problem has a feasible solution. 
In this case, the completion time of job \(J_n\) is determined by \eqref{eq:completetime1}, we have:
\begin{align*}
w_nC_n &= (2T+1)(|A_1| + rT + 1) \\
       &\leqslant (2T+1)(rT + r) \\
       &= r(T+1)(2T+1) \\
       &= Y\\
wC_{\max} &= r(T+1)(C_n + rT) = Y
\end{align*}
so $w_jC_j \leqslant wC_{\max} = Y$

If the scheduling problem exists a threshold y such that \(wC_{\text{max}} \le y\).
When $ \xi > 0 $, $C_n$is determined by \eqref{eq:completetime2}, we have:
\begin{align*}
w_nC_n &= (1 + r ( T + \xi ) + 1)(2T + 1) \\
        &\geqslant (1 + r ( T + 1 ) + 1)(2T + 1) \\
        &> Y
\end{align*}
When $ \xi < 0 $, $C_n$is determined by \eqref{eq:completetime1}, we have:
\begin{align*}
wC_{\max} &= (C_n +  \sum_{j\epsilon A_2}b_j)(T+1)  \\
         &= (1 + rT + |A_1|  + r ( T - \xi ))(T+1) \\
         &\geqslant (rT + r + rT)(T+1)  \\
         &= r(T+1)(2T+1) = Y
\end{align*}
so $wC_{\max} > Y$, there exists a contradiction.


Therefore, the Partition instance admits a solution if and only if the constructed $F2 \parallel w C_{\max}$ instance admits a schedule with $w C_{\max} \le Y$.
\hfill$\Box$



\section{\texorpdfstring{$F2 \parallel w C_{\max}$}{F2 || w C\_max} for the number of weights is constant} 
\label{sec:dp_algorithm}

When the number of distinct weights $K$ is bounded by a constant, the structural properties established in Section~\ref{sec:preliminaries} enable a polynomial-time dynamic programming algorithm.
By Lemma~\ref{lem:johnson}, the initial job sequence follows Johnson's rule; the block partition and all subsequent steps are defined with respect to this Johnson order.
We first describe the block partition and group logic underlying our approach, then present the DP formulation, and finally give the complete algorithm and its complexity analysis.

\subsection{Algorithm Properties and Dynamic Programming}

The jobs are partitioned into blocks by a single left-to-right scan. Starting from the first job, a new block is created whenever a weight value not encountered before appears; that job becomes the final job of the current block, and the next job starts a new block. Since each block introduces at least one previously unseen weight, the procedure yields at most $K$ blocks, denoted $\mathcal{B}_1, \mathcal{B}_2, \dots, \mathcal{B}_K$. By construction, the final job of each block carries the weight that triggered the cut.
By construction, block weights are non-increasing, i.e., $W(\mathcal{B}_1) \ge W(\mathcal{B}_2) \ge \cdots \ge W(\mathcal{B}_K)$. Moreover, in any optimal schedule, jobs belonging to $\mathcal{B}_1 \cup \cdots \cup \mathcal{B}_k$ must be scheduled before any job from $\mathcal{B}_{k+1} \cup \cdots \cup \mathcal{B}_K$.

Jobs of the same weight form a natural group, denoted $\mathcal{G}_1, \mathcal{G}_2, \dots, \mathcal{G}_K$. Scheduling heavier groups before lighter ones automatically enforces the non-increasing block-weight condition, and scanning groups in descending weight order directly yields the block partition.


\begin{exam}\label{ex:blocking}
Consider a 6-job instance with Johnson sequence $\pi = (J_1, \dots, J_6)$ and weights
$(w_1,\dots,w_6) = (2,\,2,\,5,\,2,\,4,\,2)$.
Scanning $\pi$ from left to right and cutting whenever a weight not seen before appears yields the following partition:
\[
\underbrace{J_1,\; J_2\; J_3}_{\mathcal{B}_1}\;
\underbrace{J_4,\; J_5}_{\mathcal{B}_2}\;
\underbrace{J_6}_{\mathcal{B}_3}
\qquad
\]
The resulting blocks are $\mathcal{B}_1=\{J_1,J_2,J_3\}$, $\mathcal{B}_2=\{J_4,J_5\}$, $\mathcal{B}_3=\{J_6\}$. And $\mathcal{G}_1=\{J_1,J_2,J_4,J_6\}$, $\mathcal{G}_2=\{J_5\}$, $\mathcal{G}_3=\{J_3\}$.
\end{exam}
\subsubsection{Dynamic Programming Formulation}

We now develop a dynamic programming algorithm that constructs the optimal schedule block by block.
Our state representation adapts the interval-based framework of (\citealp{hall1998}), originally devised for $F2|r_j|C_{\max}$, to the $w C_{\max}$ objective.

\paragraph{State Definition.}

Let $\mathscr{H}$ denote the state of $K$ blocks, represented as
\[
    \mathscr{H} =( \mathcal{B}_1, \mathcal{B}_2, \dots, \mathcal{B}_K),
\]
where each block $\mathcal{B}_k$ is described by a 4-tuple for job $J_j$, and by $L_{(k)}$ record the weight of the last job in each block :
\[
    \mathcal{B}_k = (A_j, B_j, C_j, L_{(k)}).
\]
The components are defined as follows:
\begin{itemize}
    \item $A_j$ --- Calculate the cumulative processing time on machine $M_1$;
    \item $B_j$ --- Calculate the cumulative processing time on machine $M_2$;
    \item $C_j$ --- Calculate the makespan of block $k$, computed via the recurrence of equation~\eqref{eq:makespan};
    \item $L_{(k)}$ --- the weight of the last job in block $k$.
\end{itemize}

The initial state corresponds to an empty schedule:
$\mathscr{H}_0 = (\emptyset, \dots, \emptyset)$ where each empty block is $(0, 0, 0, \emptyset)$.

\paragraph{State Transition.}

Given a state $\mathscr{H}$ after processing $k-1$ blocks for $\mathcal{J} = \{J_1, J_2, \dots, J_n\}$,
We append the next block $\mathcal{B}_k$ and $J_j$ is next job.
\begin{align}
    A_{j} &= A_{j-1} + a_j, \\
    B_{j} &= B_{j-1} + b_j, \\
    C_{j} &= \max(C_{j-1},\; A_{j}) + b_j. 
\end{align}
After all jobs in $\mathcal{B}_k$ have been processed, $L_{(k)}$ is set to $W(\mathcal{B}_k)$, the weight of the final job in block $k$, and the block 4-tuple is finalized as
\begin{equation}
    \mathcal{B}_k = (A,\; B,\; C,\; L_{(k)})
\end{equation}


\paragraph{Dominance Lemma.}

\begin{lemma}\label{lem:dominance}
Let $\mathscr{H} = (\mathcal{B}_1, \dots, \mathcal{B}_k)$ and $\mathscr{H}' = (\mathcal{B}'_1, \dots, \mathcal{B}'_k)$ be two states after processing the same set of $k$ blocks.
If $C_{(k)} \le C'_{(k)}$ and $L_{(k)} \le L'_{(k)}$, then $\mathscr{H}'$ is dominated and can be removed from further consideration.
\end{lemma}
\noindent\textbf{Proof.}
Let $\pi$ and $\pi'$ be the partial schedules corresponding to $\mathscr{H}$ and $\mathscr{H}'$.
For any suffix schedule $\sigma$ of the remaining jobs, the makespan contribution is monotone non-decreasing in the prefix completion times.
The condition $C_{(k)} \le C'_{(k)}$ ensures that $\pi$ leads to no larger makespan than $\pi'$ under any suffix.
Moreover, $L_{(k)} \le L'_{(k)}$ guarantees that the maximum weighted completion time among already scheduled jobs is no larger under $\pi$.
Hence $\pi$ dominates $\pi'$ under the $w C_{\max}$ objective.
\hfill$\Box$


\subsubsection{Algorithm 1}

The complete DP procedure is presented as three algorithms below.

\smallskip
\noindent\rule{\textwidth}{0.4pt}
\vskip -2pt
{\noindent\textbf{Block Partitioning}}
\vskip -2pt
\noindent\rule{\textwidth}{0.4pt}
\vskip 1pt
\begin{algorithmic}[1]
    \State \textbf{Input:} The Johnson sequence $\pi$ of $\mathcal{J}$; weights $w_j$, $j = 1,\dots,n$.
    \State \textbf{Output:} Blocks $\mathcal{B}_1,\dots,\mathcal{B}_K$ and their weights $W(\mathcal{B}_k)$.
    \State $S \gets \{w_{\pi(1)}\}$ 
    \State $k \gets 1$; $\mathcal{B}_k \gets \emptyset$
    \For{$i = 1$ to $n$} 
        \State $j \gets \pi(i)$
        \State $\mathcal{B}_k \gets \mathcal{B}_k \cup \{j\}$
        \If{$w_j \notin S$}
            \State $S \gets S \cup \{w_j\}$
            \State $W(\mathcal{B}_k) \gets w_j$
            \State $k \gets k + 1$; $\mathcal{B}_k \gets \emptyset$
        \EndIf
    \EndFor
    \If{$\mathcal{B}_k = \emptyset$}
        \State $K \gets k - 1$
    \Else
        \State $W(\mathcal{B}_k) \gets w_{\pi(n)}$
        \State $K \gets k$
    \EndIf
    \State \Return $\{\mathcal{B}_1, \mathcal{B}_2, \dots, \mathcal{B}_K\}$
\end{algorithmic}


\vskip -2pt
\rule{\textwidth}{0.4pt}
\vskip 4pt
{\noindent\textbf{Completion-time}}
\vskip -2pt
\noindent\rule{\textwidth}{0.4pt}
\vskip 1pt
\begin{algorithmic}[1]
    \State \textbf{Input:} Block number $K$; The job set of the first k blocks $\mathcal{J}$ with $(a_j, b_j, w_j)$; state $\mathscr{H}$.
    \State \textbf{Output:} Updated state set after processing $\mathcal{B}_k$.
    \For{$k = 1$ to $K$}
        \State $\mathcal{J}_k \gets \mathcal{B}_k$; 
        \For{each job $j \in  \mathcal{J}_k$}
            \State $A_j = A_{j-1} + a_j$
            \State $B_j = B_{j-1} + b_j$
            \State $C_j = \max(C_{j-1},\, A_{j}) + b_j$
        \EndFor
    \EndFor
    \State \Return $(A, B, C)$
\end{algorithmic}


\vskip -2pt
\rule{\textwidth}{0.4pt}
\vskip 4pt
{\noindent\textbf{dynamic programming}}
\vskip -2pt
\noindent\rule{\textwidth}{0.4pt}
\vskip 1pt
\begin{algorithmic}[1]
    \State \textbf{Input:} Jobs $\mathcal{J}$ with $(a_j, b_j, w_j)$, $j = 1,\dots,n$.
    \State \textbf{Output:} The optimal makespan $C_{\max}$.
    \State $\{\mathcal{B}_1, \mathcal{B}_2, \dots, \mathcal{B}_K\} \gets$ \textbf{Block Partitioning}($\pi$)
    \State $\mathscr{H} \gets \{ (0,\, 0,\, 0,\, 0) \}$
    \For{$k = 1$ to $K$}
        \State $\mathscr{H} \gets \textbf{Completion-time}(k,\, \mathscr{H})$
        \For{each state $s \in \mathscr{H}$}
            \State $L_{(k)} \gets W(\mathcal{B}_k)$
        \EndFor
    \EndFor
    \State $C_{\max} \gets \min\{\, C \mid (A,\, B,\, C,\, L) \in \mathscr{H} \,\}$
    \State \Return $C_{\max}$
\end{algorithmic}
