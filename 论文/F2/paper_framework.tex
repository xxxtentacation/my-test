\begin{abstract}
    Minimizing $W C_{\max}$ on a two-machine flow shop,
    denoted $F2 \parallel W C_{\max}$, is a classical NP-hard scheduling problem.
    Existing exact methods rely on enumerative techniques that scale poorly with instance size,
    while approximation algorithms often impose restrictive assumptions on processing times.
    We first sequence all jobs globally by Johnson's rule, obtaining an initial schedule $\pi$.
    Scanning $\pi$ from left to right, we partition it into at most $K$ blocks: whenever a weight not seen before appears, 
    it serves as the final job of the current block, and the next job starts a new block.
    We establish structural properties of optimal schedules: the permutation property, the positional constraint that block $\mathcal{G}_k$ jobs cannot appear beyond the first $k$ blocks,
    and the block-end condition requiring the final job of each block to carry that block's weight (automatically satisfied by construction).
    The intra-block Johnson ordering inherited from $\pi$ is further formalized within the DP framework.
    Building on these properties and the interval-based dynamic programming framework (\citealp{hall1994}) developed for $F2|r_j|C_{\max}$,
    we design a polynomial-time dynamic programming algorithm with $O(|Y|_{\max}\cdot n)$ complexity that appends jobs block by block.
    We further show that geometric rounding of weights and processing times converts this algorithm into a polynomial-time approximation scheme.
\end{abstract}

\noindent \textbf{Keywords:} Flow shop scheduling; $W C_{\max}$; Dynamic programming; Structural properties

\section{Preliminaries}
\label{sec:preliminaries}

We consider the problem $F2 \parallel W C_{\max}$.
Let $\mathcal{J} = \{J_1, J_2, \dots, J_n\}$ be the set of $n$ jobs to be processed on two machines $M_1$ and $M_2$ in series:
 each job $J_j$ must be processed first on $M_1$ for $a_j$ time units and then on $M_2$ for $b_j$ time units, with no preemption allowed.
All processing times $a_j$ and $b_j$ are assumed to be positive integers.
$M_1$ and $M_2$ can process at most one job at a time.
Each job $J_j$ has a weight $w_j > 0$.
A schedule $\pi = (\pi(1), \pi(2), \dots, \pi(n))$ specifies the processing order on both machines.
For a schedule $\pi$, let $A_j = \sum_{h=1}^{j} a_{\pi(h)}$ and $B_j = \sum_{h=1}^{j} b_{\pi(h)}$ 
be the cumulative $M_1$ and $M_2$ completion times after the first $j$ jobs.
The objective is to find a schedule $\pi$ that minimizes $W C_{\max}$.


The problem $F2 \parallel W C_{\max}$ can be formulated as a mixed integer linear program (MILP).
For a permutation schedule, let the positions be indexed by $k = 1,\dots,n$.
We introduce the following decision variables:
\begin{itemize}
	\item $x_{jk} = 1$ if job $J_j$ processing at position $k$, and $0$ otherwise;
	\item $C_k^1$ completion time of the job at position $k$ on $M_1$;
	\item $C_k^2$ completion time of the job at position $k$ on $M_2$;
	\item $C_j$ completion time of job $J_j$ on $M_2$;
	\item $Z$ the maximum weighted completion time $W C_{\max}$.
	\item $M = 2\sum_{j=1}^{n}(a_j + b_j)$ be a sufficiently large constant.
\end{itemize}
The MILP model is:
\begin{align}
	\min \quad & Z \label{milp:obj} \\[4pt]
	\text{s.t.} \quad & \sum_{k=1}^{n} x_{jk} = 1, \quad \forall\, j \in \mathcal{J} \label{milp:assign1} \\[4pt]
	& \sum_{j=1}^{n} x_{jk} = 1, \quad \forall\, k = 1,\dots,n \label{milp:assign2} \\[4pt]
	& C_1^1 = \sum_{j=1}^{n} a_j x_{j1} \label{milp:m1_start} \\[4pt]
	& C_k^1 = C_{k-1}^1 + \sum_{j=1}^{n} a_j x_{jk}, \quad \forall\, k = 2,\dots,n \label{milp:m1_rec} \\[4pt]
	& C_1^2 = C_1^1 + \sum_{j=1}^{n} b_j x_{j1} \label{milp:m2_start} \\[4pt]
	& C_k^2 \ge C_k^1 + \sum_{j=1}^{n} b_j x_{jk}, \quad \forall\, k = 2,\dots,n \label{milp:m2_a} \\[4pt]
	& C_k^2 \ge C_{k-1}^2 + \sum_{j=1}^{n} b_j x_{jk}, \quad \forall\, k = 2,\dots,n \label{milp:m2_b} \\[4pt]
	& C_j \ge C_k^2 - M(1 - x_{jk}), \quad \forall\, j \in \mathcal{J},\; k = 1,\dots,n \label{milp:link} \\[4pt]
	& Z \ge w_j C_j, \quad \forall\, j \in \mathcal{J} \label{milp:wcmax} \\[4pt]
	& x_{jk} \in \{0,1\},\quad C_k^1, C_k^2, C_j, Z \ge 0 \label{milp:dom}
\end{align}

Constraint (\ref{milp:assign1})--(\ref{milp:assign2}) ensure a one-to-one assignment between jobs and positions.
Constraints (\ref{milp:m1_start})--(\ref{milp:m1_rec}) compute the completion time of each position on $M_1$.
Constraints (\ref{milp:m2_start})--(\ref{milp:m2_b}) compute the completion time of each position on $M_2$:
for $k \ge 2$, the job at position $k$ starts on $M_2$ only after it completes $M_1$ and the preceding job releases $M_2$.
Constraint (\ref{milp:link}) links the position-based completion times to the job-based completion times via a big-$M$ formulation:
when $x_{jk} = 1$, job $J_j$ is at position $k$ and $C_j \ge C_k^2$;
when $x_{jk} = 0$, the right-hand side becomes $C_k^2 - M \le 0$, which is non-binding.
Constraint (\ref{milp:wcmax}) enforces $Z \ge w_j C_j$ for every job, so that minimizing $Z$ yields the optimal $W C_{\max}$.





\section{Problem formulation}
\begin{theorem}\label{thm:nphard}
    $F2 \parallel W C_{\max}$ is NP-hard.
\end{theorem}
\begin{proof}
We reduce from the Partition problem: given $n$ positive integers $x_1, x_2, \dots, x_r$ with $\sum_{i=1}^{r} x_i = 2T$, 
does there exist subset $Z_1$ and $Z_2 \subseteq \{1,\dots,r\}$ such that:
 $\sum_{i \in Z_1} x_i$ = $\sum_{i \in Z_2} x_i$ = $T$?

Construct an instance of $F2 \parallel W C_{\max}$,  Let the number of jobs $n = r + 1$.
Let $a_j = 1$, $b_j = rx_j$, $w_j = T+1$ for jobs $j = 1,\dots,n-1$.
Put $a_n = rT$, $b_n = 1$, $w_n = 2T+1$ for job $J_n$.
Does there exist a threshold Y such that:
\[
W C_{\max} \le Y = r(T+1)(2T+1)
\]
holds?

We partition jobs $j_1$ to $j_{n-1}$ into two subsets via $j_n$, 
the distribution of jobs on the machine is shown in Figure~\ref{fig:3-1}.
Since the processing times of $A_1$ and $A_2$ are unknown, while their total processing time is 2T. 
We may assume that the processing time of $A_1$ on $M_2$ is $r(T+\xi)$, $A_2$ is $r(T-\xi)$, where $-T\le\xi\le T$.
Therefore, the completion time of $J_n$ is determined by the following formula:
\begin{equation}\label{eq:completetime1}
    \qquad
    C_n^1 = |A_1| + rT + 1
\end{equation}
\begin{equation}\label{eq:completetime2}
    \qquad
    C_n^2 = 1 + r ( T + \xi ) + 1
\end{equation}
and $C_n = \max \{C_n^1, C_n^2\}$

If $\xi = 0$, then $\sum_{j\epsilon A_1}b_j =\sum_{j\epsilon A_2}b_j $, which means the partition problem has a feasible solution. 
In this case, the completion time of job \(J_n\) is determined by \eqref{eq:completetime1}, we have:
\begin{align*}
C_n &= |A_1| + rT + 1 \leqslant (rT + r) = r(T+1) \\
w_nC_n &= (2T + 1)(r(T+1)) = Y\\
W C_{\max} &= r(T+1)(C_n + rT) = Y\\
\end{align*}
so $w_jC_j \leqslant W C_{\max} = Y$

If the scheduling problem exists a threshold y such that \(W C_{\text{max}} \le y\).
When $ \xi > 0 $, $C_n$is determined by \eqref{eq:completetime2}, we have:
\begin{align*}
C_n &= (1 + r ( T + \xi ) + 1) \geqslant (1 + r ( T + 1 ) + 1) \\
w_nC_n  &\geqslant (2T + 1)(1 + r ( T + 1 ) + 1) = Y
\end{align*}
When $ \xi < 0 $, $C_n$is determined by \eqref{eq:completetime1}, we have:
\begin{align*}
C_{\max} &= (C_n +  \sum_{j\epsilon A_2}b_j)  \\
         &= (1 + rT + |A_1|  + r ( T - \xi )) \\
         &\geqslant (rT + r + rT)  \\
         &= r(2T+1) = Y \\
WC_{\max} &\geqslant r(T+1)(r(2T+1)) = Y\\
\end{align*}
so $W C_{\max} > Y$, there exists a contradiction.

Therefore, the Partition instance admits a solution if and only if 
the constructed $F2 \parallel W C_{\max}$ instance admits a schedule with $W C_{\max} \le Y$.
\end{proof}

\begin{theorem}\label{thm:snphard}
    $F2 \parallel W C_{\max}$ is strongly NP-hard.
\end{theorem}
\begin{proof}
We reduce from the 3-Partition problem: given $3m$ positive integers $x_1,\dots,x_{3m}$ and a bound $B$ with
$B/4 < x_i < B/2$ and $\sum_{i=1}^{3m} x_i = mB$, can the integers be partitioned into $m$ triples each summing to $B$?
Construct an instance of $F2 \parallel W C_{\max}$ with $n = 4m$ jobs as follows
(see Figure~\ref{fig:3-2}):
\begin{itemize}
  \item $3m$ element jobs $J_1,\dots,J_{3m}$: $a_j = 1$, $b_j = x_j + \frac{2}{3} $, $w_j = 1$;
  \item $m$ separator jobs $S_1,\dots,S_{m}$: $a_{s_k} = B$, $b_{s_k} = 0$, $w_{s_k} = 3m/k$.
\end{itemize}
The total $M_1$ work is $\sum a_j = 3m + mB = m(B+3)$,
the total $M_2$ work is $\sum b_j = \sum_{j=1}^{3m}(x_j + \frac{2}{3}) = mB + 2m = m(B+2)$,
and the threshold is set to $Y = 3m(B+3)$.

Suppose 3-Partition exists. There is a partition of the $3m$ elements into
$m$ triples $G_1,\dots,G_m$ with $\sum_{J_j \in G_k} x_j = B$ for each $k$.
Arrange the jobs in the order $G_1, S_1, G_2, S_2, \dots, G_m, S_m$.
On $M_1$, the three elements of $G_k$ each consume $1$ unit ($3$ in total), after which $S_k$ occupies $B$ units;
thus $S_k$ completes $M_1$ at time $k(B+3)$.
On $M_2$, the three elements of $G_k$ have total processing time $\sum (x_j+\frac{2}{3}) = B+2$.
The first element of $G_1$ finishes $M_1$ at time $1$ and starts $M_2$ immediately,
finishing $M_2$ at $1 + (x_1+\frac{2}{3})$.
Since each $x_j > B/4$ is large relative to the unit $M_1$ times,
the $M_2$ pipeline is the bottleneck, and the last element of $G_1$ completes $M_2$ at $B+3$.
$S_1$ has $b = 0$ and passes $M_2$ instantly at $B+3$.
Inductively, $S_k$ completes $M_2$ at $C_{s_k} = k(B+3)$.
Consequently
\[
w_{s_k} C_{s_k} = \frac{3m}{k} \cdot k(B+3) = 3m(B+3) = Y,
\]
while each element job has $w_j = 1$ and completion time at most $m(B+3)$, giving
$w_j C_j \le m(B+3) \le Y$ for all $m \ge 3$.
Hence $W C_{\max} = Y$.

Conversely, suppose there exists a schedule $\pi$ with $W C_{\max} \le Y = 3m(B+3)$.
For separator $S_k$ we have $w_{s_k} C_{s_k} = \frac{3m}{k} C_{s_k} \le 3m(B+3)$, i.e.,
$C_{s_k} \le k(B+3)$.
Because each separator requires $B$ time on $M_1$, the $k$-th separator in schedule order
cannot finish $M_2$ before $M_1$ has processed at least $k$ separators
plus all element jobs preceding them.
Let $E_k$ be the number of element jobs preceding $S_k$, and let $e_k = E_k - E_{k-1}$ (with $E_0 = 0$)
be the number of element jobs between $S_{k-1}$ and $S_k$.
From $C_{s_k} \ge kB + E_k$ and $C_{s_k} \le k(B+3)$, we obtain $E_k \le 3k$.
Since $E_m = 3m$, each inequality must hold with equality: $E_k = 3k$ for all $k$,
hence $e_k = 3$ for every block.
Denote the three elements between $S_{k-1}$ and $S_k$ by $G_k$, with
$\sum_{J_j \in G_k} x_j = B_k$.

Now analyze the $M_2$ pipeline of block $k$. The three elements of $G_k$ have total $M_2$ work
$B_k + 2$ and unit $M_1$ time each.
From the flow-shop critical-path formula applied to $G_k$ followed by $S_k$,
the span satisfies $C_{s_k} \ge C_{s_{k-1}} + \max\{B_k+3,\; B+3\}$
(the $B_k+3$ term arises from the first element starting $M_2$ one unit after the block begins,
followed by $B_k+2$ total $M_2$ work on the three elements of $G_k$,
while $B+3$ is the pure $M_1$ path through $S_k$).
Since $C_{s_k} \le C_{s_{k-1}} + B + 3$ (otherwise $C_{s_k}$ exceeds $k(B+3)$
given $C_{s_{k-1}} \le (k-1)(B+3)$), we obtain
\[
B_k + 3 \;\le\; C_{s_k} - C_{s_{k-1}} \;\le\; B + 3
\quad\Longrightarrow\quad
B_k \;\le\; B.
\]
Summing over all $m$ groups yields $\sum_{k=1}^m B_k \le mB$,
but the total of all $x_j$ equals $mB$ by construction; therefore each inequality holds with equality:
$\sum_{J_j \in G_k} x_j = B$ for every $k$.
With $B/4 < x_j < B/2$, three elements can sum to exactly $B$ only if
each group is a valid triple; fewer than three would sum to ${}<B$, more than three to ${}>B$.
Hence $\{G_1,\dots,G_m\}$ is a partition of the $3m$ elements into $m$ triples each summing to $B$.

Thus $F2 \parallel W C_{\max}$ is strongly NP-hard.
\end{proof}

\section{Algorithm} 
\label{sec:dp_algorithm}
In this chapter, we will propose an idea to guide the dynamic programming algorithm for solving this problem, 
and present the time complexity analysis of the algorithm.

\subsection{The number of weights is constant}
For $F2 \parallel W C_{\max}$, a permutation schedule in which the job order on $M_1$ and $M_2$, 
Possess an identical processing order for jobs (\citealp{baker1974introduction}).
Johnson's rule (\citealp{johnson1954optimal}) sequences all $n$ jobs to minimize the makespan:
jobs with $a_j \le b_j$ go first in non-decreasing order of $a_j$, and the remaining jobs follow
in non-increasing order of $b_j$.

For any scheduling $\pi$. Let $W_{1}, W_{2}, \dots, W_{K}$ be the $K$ distinct weights among all jobs, 
and $W_{1} > W_{2} > \dots > W_{K}$ and define $\mathcal{G}_k = \{ J_j \in \mathcal{J} \mid w_j = W_{k} \}$, $k = 1,\dots,K$.
For each group $\mathcal{G}_k$, let $\ell_k = \arg\max \{\, p : \pi(p) \in \mathcal{G}_k \,\}, {p \in \{1,\dots,n\}}$, and set $\ell_0 = 0$;
The blocks are defined as: 
For $J_j \in \mathcal{J}$, $\mathcal{B}_1 = \{J_1, \dots, J_{\ell_1}\}$, $\mathcal{J} = \{\mathcal{J} \setminus \mathcal{B}_1 \}$; 
$\mathcal{B}_2 = \{J_{\ell_1}, \dots, J_{\ell_2}\}$, $\mathcal{J} = \{\mathcal{J} \setminus \mathcal{B}_2\}$; 
$\mathcal{B}_k = \{J_{\ell_{k-1}}, \dots, J_{\ell_k}\}$.

Note a block is allowed empty and accepts any job. In particular, if $\ell_k = \ell_{k-1}$, then
$\mathcal{B}_k = \{J_{\ell_{k-1}}, \dots, J_{\ell_k}\}$ contains no job and is called an empty block;
an empty block $\mathcal{B}_i$ has $A_i = B_i = C_i = 0$.



$A_i$ total processing time on machine $M_1$ of jobs in block $\mathcal{B}_i$;
$B_i$ total processing time on machine $M_2$ of jobs in block $\mathcal{B}_i$;
$C_i$ completion time of block $\mathcal{B}_i$ on machine $M_2$;
$L_i$ weight of the last job in block $\mathcal{B}_i$.
Thus each block weights are non-increasing: $W_{\mathcal{B}_1} \ge W_{\mathcal{B}_2} \ge \cdots \ge W_{\mathcal{B}_K}$.

For ease of explanation, we construct an illustrative schedule.

\begin{lemma}\label{lem:block-johnson}
For any schedule $\pi$, reordering the jobs within each block according to Johnson's rule~\citealp{johnson1954optimal}
and concatenating the resulting block schedules yields a schedule $\pi'$ with $W C_{\max}(\pi') \le W C_{\max}(\pi)$.
\end{lemma}
\begin{proof}
By construction, the final job of block $\mathcal{B}_k$ carries weight $W(\mathcal{B}_k)$,
and every job in $\mathcal{B}_k$ has weight at most $W(\mathcal{B}_k)$:
a weight occurring in $\mathcal{B}_k$ either equals $W(\mathcal{B}_k)$ or has already appeared in an earlier block with larger weight.
Let $C_k(\sigma)$ denote the completion time of the last job of $\mathcal{B}_k$ under schedule $\sigma$.

We first argue that within each block, the maximum weighted completion time is attained at its last job.
For any $J_j \in \mathcal{B}_k$, we have $C_j(\sigma) \le C_k(\sigma)$ because the last job finishes no earlier than any other job in the block,
and $w_j \le W(\mathcal{B}_k)$ by the block construction.
Hence $w_j C_j(\sigma) \le W(\mathcal{B}_k) \cdot C_k(\sigma)$.
Taking the maximum over $j \in \mathcal{B}_k$ yields
$\max_{J_j \in \mathcal{B}_k} w_j C_j(\sigma) = W(\mathcal{B}_k) \cdot C_k(\sigma)$,
and therefore
\[
W C_{\max}(\sigma) = \max_{k=1,\dots,K}\{W(\mathcal{B}_k) \cdot C_k(\sigma)\}.
\]
Thus the overall objective depends only on the last job of each block.

Now let $\pi'$ be the schedule obtained by reordering each $\mathcal{B}_k$ internally via Johnson's rule,
preserving the block order. We prove $C_k(\pi') \le C_k(\pi)$ for all $k$ by induction.

\textbf{Base case $k=1$.}
Block $\mathcal{B}_1$ starts at time zero on both machines regardless of internal ordering,
so Johnson's rule directly gives $C_1(\pi') \le C_1(\pi)$.

\textbf{Inductive step.}
Assume $C_{k-1}(\pi') \le C_{k-1}(\pi)$.
On $M_1$, the start time of $\mathcal{B}_k$ equals the total $M_1$ work of $\mathcal{B}_1,\dots,\mathcal{B}_{k-1}$,
which is invariant under internal reordering; thus $\mathcal{B}_k$ begins at the same instant on $M_1$ under both schedules.
On $M_2$, $\mathcal{B}_k$ starts at $\max\{\,\sum_{i=1}^{k}A_{i},\; C_{k-1}\,\}$,
The first term is identical under $\pi$ and $\pi'$, and the second is no larger under $\pi'$ by the induction hypothesis.
Hence $\mathcal{B}_k$ starts no later on $M_2$ under $\pi'$ than under $\pi$.
With the same $M_1$ start and an internally optimal Johnson ordering,
we obtain $C_k(\pi') \le C_k(\pi)$, completing the induction.

Finally,
\[
W C_{\max}(\pi') = \max_{k} \; W(\mathcal{B}_k) \cdot C_k(\pi')
   \le \max_{k} \; W(\mathcal{B}_k) \cdot C_k(\pi)
   = W C_{\max}(\pi),
\]
which establishes the lemma.
\end{proof}


\subsection{Algorithm Properties and Dynamic Programming}
We now develop a dynamic programming algorithm that processes jobs sequentially in Johnson order
and assigns each job to one of the $K$ blocks $\mathcal{B}_1,\dots,\mathcal{B}_K$.

Lemma~\ref{lem:block-johnson} provides the structural basis:
it guarantees that, given any assignment of jobs to the $K$ blocks,
sequencing the jobs within each block $\mathcal{B}_i$ according to Johnson's rule
and concatenating the blocks in non-increasing weight order yields a schedule
whose makespan is no larger than that of any other schedule respecting the same assignment.
For each job $J_j \in \mathcal{J}$ ($j = 1,\dots,n$), let
\[
    \mathcal{H}_j = \{\, A_i; \; B_i, C_i; \; L_i \,\}
\]
denote the state associated with $J_j$, where the index $i$ ranges over the blocks eligible for $J_j$,
and in every block preceding it:
\begin{itemize}
    \item $A_i$ --- total processing time on machine $M_1$ of jobs in block $\mathcal{B}_i$;
    \item $B_i$ --- total processing time on machine $M_2$ of jobs in block $\mathcal{B}_i$;
    \item $C_i$ --- completion time of the last job of block $\mathcal{B}_i$ on machine $M_2$;
    \item $L_i$ --- weight of block $\mathcal{B}_i$.
\end{itemize}
Since the block weights are non-increasing, $W_{1} > \cdots > W_{K}$, the blocks eligible for $J_j$
form a prefix:
\[
    \mathcal{E}(J_j) = \{\, i : w_j \le W_i \,\}, \qquad
    i^* = \arg\max_{i \in \{1,\dots,K\}} \{\, i : W_i \ge w_j \,\},
\]
so that $\mathcal{E}(J_j) = \{\, 1, \dots, i^* \,\}$.
A state is in fact the aggregate $K$-block tuple, described by $4K-2$ parameters: $A_K$ and $B_K$ need not be stored,
since after the first $j$ jobs have been processed, every job belongs to exactly one block and the total
processing times $\sum_{i=1}^{K} A_i = \sum_{j=1}^{n} a_{\pi(j)}$ and $\sum_{i=1}^{K} B_i = \sum_{j=1}^{n} b_{\pi(j)}$ are known,
so $A_K$ and $B_K$ are recovered by
\[
    A_K = \sum_{j=1}^{n} a_{\pi(j)} - \sum_{i=1}^{K-1} A_i, \qquad
    B_K = \sum_{j=1}^{n} b_{\pi(j)} - \sum_{i=1}^{K-1} B_i.
\]
Hence a state is represented as
\[
    (C_1,\dots,C_K; \; A_1,\dots,A_{K-1}; \; B_1,\dots,B_{K-1}; \; L_1,\dots,L_K) \in \mathscr{H}_j.
\]


The initial state set contains a single state with all components set to zero:
$\mathscr{H}_{0} = \{ (0,\dots,0; \; 0,\dots,0; \; 0,\dots,0; \; 0,\dots,0) \}$.

\begin{lemma}\label{lem:merge-dominance}
Let $H = (C_1,\dots,C_K; A_1,\dots,A_{K-1}; B_1,\dots,B_{K-1}; L_1,\dots,L_K)$ and
$H' = (C'_1,\dots,C'_K; A'_1,\dots,A'_K; B'_1,\dots,B'_K; L'_1,\dots,L'_K)$
be two states in $\mathscr{H}_j$ for some $j$.
If $A_i = A'_i$, $B_i = B'_i$, $L_i = L'_i$ for all $i$, and $C_i \le C'_i$ for all $i$,
then $H'$ can be discarded without affecting the optimal solution.
\end{lemma}
\begin{proof}
Consider any continuation of a schedule extending state $H'$ to a complete schedule $\Pi'$.
Replacing the prefix represented by $H'$ with the prefix represented by $H$ yields a schedule $\Pi$
that uses the same assignment of the remaining jobs.
For every subsequent block $\mathcal{B}_k$ ($k \ge 2$), the $M_2$ start time is
$\max\{\sum_{i=1}^{k}A_{i},\, C_{k-1}\}$, while $C_{k-1}$ under $H$ is no larger than under $H'$.
By induction, the $M_2$ completion time of every subsequent block is no larger under $H$,
and since $A_i$, $B_i$, $L_i$ are identical, the objective value $W C_{\max}$ of $\Pi$ does not exceed that of $\Pi'$.
\end{proof}

During the merge step, the algorithm applies this dominance rule:
among states with identical $(A_i, B_i, L_i)_{i=1}^K$, only those with component-wise minimal $(C_i)_{i=1}^K$ are retained.

For $j = 1, \dots, n$, the algorithm extends $\mathscr{H}_{j-1}$ to $\mathscr{H}_j$ as follows.
For each state in $\mathscr{H}_{j-1}$ and each block $i \in \mathcal{E}(J_j)$ (every block that precedes the block of $J_j$, i.e., every $i$ with $w_j \leqslant L_i$),
the algorithm places $J_j$ at the end of $\mathcal{B}_i$. Only the $i$-th slice of the state is updated, while all other blocks remain unchanged:
\[
C'_i = \max\{C_i + b_j,\; A_i + a_j + b_j\},\quad
A'_i = A_i + a_j,\quad
B'_i = B_i + b_j.
\]
The resulting candidate states are collected in a temporary set $\mathcal{Y}$ (the sub-states).
Via Lemma~\ref{lem:merge-dominance}, merging the candidates removes dominated and duplicate states and yields $\mathscr{H}_{j}$.


Once all jobs are assigned and each block $\mathcal{B}_i$ is summarized by its parameters $(A_i, B_i, C_i)$,
the overall makespan $C_{\max}$ is obtained by concatenating the blocks in order of increasing index.
The procedure is given below:

In the update of $T_B$,
the term $T_A - A_i$ is the start time of $\mathcal{B}_i$ on $M_1$, so $T_A - A_i + C_i$
is the completion time of $\mathcal{B}_i$ on $M_2$ when $M_2$ is free,
while $T_B + B_i$ accounts for the case where $M_2$ is the bottleneck.

\setlength{\intextsep}{4pt}
\begin{algorithm}[H]
\caption{Block concatenation and evaluation}
\label{alg:Block-concat}
\begin{algorithmic}[1]
    \State \textbf{Input:} States $\mathscr{H}$.
    \State \textbf{Output:} $W C_{\max} = \max_i\, L_i \cdot T_B^{(i)}$.
    \State $T_{A_0} \gets 0$, \enspace $T_{B_0} \gets 0$, \enspace $W C_{\max} \gets 0$
    \For{$i = 1$ to $K$}
        \State $T_{A_i} \gets T_{A_{i-1}} + A_i$
        \State $T_{B_i} \gets \max\{\, T_{A_{i-1}} + C_i,\; T_{B_{i-1}} + B_i \,\}$
        \State $W C_{\max} \gets \max\{\, W C_{\max},\; L_i \cdot T_{B_i} \,\}$
    \EndFor
    \State \Return $W C_{\max}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{DP for $F2 \parallel W C_{\max}$}
\label{alg:dp}
\begin{algorithmic}[1]
    \State \textbf{Input:} Jobs $\mathcal{J}$ with $(a_j, b_j, w_j)$ in Johnson order; block count $K$; distinct weights $W_{1}>\cdots>W_{K}$.
    \State \textbf{Output:} The optimal value $Z^* = \min_S \max_i\,W_{i}\cdot T_B^{(i)}$.
    \State Set $L_i \gets W_{i}$ for $i=1,\dots,K$
    \State $\mathscr{H}_{0} \gets \{ (\; 0,\dots,0; \; 0,\dots,0; \; 0,\dots,0; \; 0,\dots,0) \}$
    \For{$j = 1$ to $n$}
        \State $\mathcal{Y}_i \gets \emptyset$ for $i = 1,\dots,K$
        \For{each $(\{C_i,A_i,B_i,L_i\}_{i=1}^K) \in \mathscr{H}_{j-1}$}
            \For{$i = 1$ to $K$}
                \If{$w_j \le L_i$}
                    \State $C_i' \gets \max\{\, C_i + b_j,\; A_i + a_j + b_j \,\}$
                    \State $A_i' \gets A_i + a_j$, \enspace $B_i' \gets B_i + b_j$
                    \State Add the updated state to $\mathcal{Y}_i$
                \EndIf
            \EndFor
        \EndFor
        \State $\mathscr{H}_{j} \gets \textsc{Merge}(\mathcal{Y}_1,\dots,\mathcal{Y}_K)$
    \EndFor
    \State $Z^* \gets +\infty$
    \For{each state $S = \{C_i, A_i, B_i, L_i\}_{i=1}^K \in \mathscr{H}_{n}$}
        \State $W C_{\max} \gets \textsc{Algorithm~\ref{alg:Block-concat}}\bigl((A_i,B_i,C_i,L_i)_{i=1}^K\bigr)$
        \State $Z^* \gets \min\{\, Z^*,\; W C_{\max} \,\}$
    \EndFor
    \State \Return $Z^*$
\end{algorithmic}
\end{algorithm}

\begin{theorem}\label{thm:complexity}
Algorithm~2 solves $F2 \parallel W C_{\max}$ optimally in
$O\bigl(n \cdot K^2 \cdot |\mathscr{H}|_{\max}\bigr)$ time, where
$|\mathscr{H}|_{\max} \le K^K \cdot V^{\,3K-3}$,
$V = \max\bigl\{\sum_{j=1}^n a_j,\;\sum_{j=1}^n b_j\bigr\}$.
For constant $K$, this is $O(n \cdot V^{\,3K-3})$.
\end{theorem}

\begin{proof}
Each state in $\mathscr{H}_{j}$ is a $(4K-2)$-tuple
$(C_1,\dots,C_K;\, A_1,\dots,A_{K-1};\, B_1,\dots,B_{K-1};\, L_1,\dots,L_K)$.
We bound the number of distinct values each component can attain.

The quantity $A_i$ is the total $M_1$ processing time of jobs assigned to block $\mathcal{B}_i$,
hence $A_i \in [0,\,V]$.
Since the job sequence is fixed and the $j$ jobs under consideration form a prefix of the Johnson order,
$\sum_{i=1}^K A_i$ equals the sum of $a_j$ over that prefix.
Consequently at most $K-1$ of the $A_i$ are independent, contributing at most $V^{\,K-1}$ distinct $A$-vectors.
Symmetrically, $\sum_{i=1}^K B_i$ is the prefix sum of the $b_j$'s,
so the $B_i$ components likewise contribute at most $V^{\,K-1}$ distinct combinations.

The quantity $C_i$ is the makespan of $\mathcal{B}_i$ when scheduled in isolation by Johnson's rule.
Since $C_i \le A_i + B_i \le 2V$, the $C$-vector contributes at most $(2V)^K$ distinct values.
The weight $L_i$ is the weight of the last job in $\mathcal{B}_i$ and takes values from $\{w^{(1)},\dots,w^{(K)}\}$,
yielding at most $K^K$ distinct $L$-vectors.

Multiplying these bounds yields
\[
|\mathscr{H}_{j}| \le K^K \cdot (2V)^{K-1} \cdot V^{\,2K-2}
                      = O\bigl(K^K \cdot 2^K \cdot V^{\,3K-3}\bigr).
\]

In iteration $j$, the algorithm takes each state in $\mathscr{H}_{j-1}$, tries every
block $i \in \{1,\dots,K\}$, computes the updated tuple in $O(1)$ time,
and inserts it into $Y_i$.
The merge step collects the $K$ sets $Y_1,\dots,Y_K$ and removes duplicates in time
proportional to the total number of generated tuples.
Hence iteration $j$ costs $O(K \cdot |\mathscr{H}_{j-1}|)$,
and summing over all $n$ iterations gives $O(n \cdot K \cdot |\mathscr{H}|_{\max})$.

The final concatenation via Algorithm~1 evaluates each state in $\mathscr{H}_{n}$
in $O(K)$ time, which is dominated by the DP phase.
When $K$ is fixed, $K^K \cdot 2^K$ is constant, giving $O(n \cdot V^{\,3K-3})$.
\end{proof}






\subsection{FPTAS}
\label{sub:fptas}

We turn the exact dynamic program of Algorithm~\ref{alg:dp} into a fully
polynomial-time approximation scheme by geometric rounding of its state space.
Only the completion-time coordinates $A_i,B_i,C_i$ are rounded. 

Fix $0<\varepsilon<1$ and let
\[
\delta=1+\frac{\varepsilon}{2n},\qquad U=2V,\qquad v=\Big\lceil\log_\delta U\Big\rceil,
\]
where $V=\max\bigl\{\sum_{j=1}^n a_j,\ \sum_{j=1}^n b_j\bigr\}$.
Partition $[0,U]$ into the $v+1$ intervals
\begin{equation}\label{eq:fptas-intervals}
I_0=[0,1),\qquad I_r=[\delta^{r-1},\delta^{r})\ \ (1\le r\le v-1),\qquad
I_v=[\delta^{v-1},U].
\end{equation}
All processing times are assumed integral, so every value taken by a coordinate
$A_i,B_i,C_i$ is a nonnegative integer; hence any two values lying in a common
interval satisfy $x\le \delta\,y$. We call this the \emph{interval-dominance}
property; it is the only fact about the rounding used below.

A \emph{box} is the set of states whose time coordinates fall in a fixed tuple
of intervals and whose weight labels are fixed. For
$\mathbf r^{A},\mathbf r^{B},\mathbf r^{C}\in\{0,\dots,v\}^{K}$ and
$\mathbf L=(L_1,\dots,L_K)\in\big(\{0\}\cup\{W_1,\dots,W_K\}\}\big)^K$,
where $L_i=0$ marks an empty block $\mathcal{B}_i$, the box is
\[
\mathscr{B}(\mathbf r^{A};\mathbf r^{B};\mathbf r^{C};\mathbf L)
=\Big\{\,(C_i,A_i,B_i,L_i)_{i=1}^K\;:\;
A_i\in I_{r_i^{A}},\ B_i\in I_{r_i^{B}},\ C_i\in I_{r_i^{C}}\Big\}.
\]
There are $(v+1)^{3K}(K+1)^K$ boxes, a number independent of the job sizes.

\begin{algorithm}[H]
\caption{Boxed FPTAS for $F2 \parallel W C_{\max}$}
\label{alg:fptas}
\begin{algorithmic}[1]
    \State \textbf{Input:} Jobs $\mathcal{J}$ with $(a_j,b_j,w_j)$ in Johnson order; block count $K$; accuracy $\varepsilon$.
    \State \textbf{Output:} A schedule with $W C_{\max}\le(1+\varepsilon)\,W C_{\max}^{*}$.
    \State Set $\delta=1+\varepsilon/(2n)$, $U=2V$, $v=\lceil\log_\delta U\rceil$, and the intervals of \eqref{eq:fptas-intervals}
    \State Store the zero state as the representative of $\mathscr{B}(\mathbf 0;\mathbf 0;\mathbf 0;\mathbf 0)$
    \For{$j=0$ to $n-1$}
        \For{each box $\mathscr{B}$ with representative $s=(C_i,A_i,B_i,L_i)_{i=1}^K$}
            \For{$i=1$ to $K$}
                \If{block $i$ is empty \textbf{or} $w_{j+1}\le L_i$}
                    \State Compute the candidate $\tilde s$ by the recurrences of Algorithm~\ref{alg:dp} (append $J_{j+1}$ to block $i$; 
                            if block $i$ was empty, set $L_i\gets w_{j+1}$)
                    \State Map $\tilde s$ to its box $\mathscr{B}'$
                    \If{$\mathscr{B}'$ has no representative} Store $\tilde s$ and record $s$ as its predecessor
                    \Else\ Keep either one of the two representatives of $\mathscr{B}'$ \EndIf
                \EndIf
            \EndFor
        \EndFor
    \EndFor
    \State Evaluate every representative at level $n$ with Algorithm~\ref{alg:Block-concat}
    \State \Return the schedule attaining the minimum $W C_{\max}$
\end{algorithmic}
\end{algorithm}

\begin{theorem}\label{thm:fptas-ratio}
For every $0<\varepsilon<1$, Algorithm~\ref{alg:fptas} returns a schedule with
\[
W C_{\max}\ \le\ (1+\varepsilon)\,W C_{\max}^{*},
\]
where $W C_{\max}^{*}$ is the optimal value of $F2 \parallel W C_{\max}$.
\end{theorem}

\begin{proof}
Let $s_0^{*}\to s_1^{*}\to\cdots\to s_n^{*}$ be a trajectory of
Algorithm~\ref{alg:dp} that ends in an optimal schedule, and write
$s_j^{*}=(C_i^{j*},A_i^{j*},B_i^{j*},L_i^{j*})_{i=1}^K$.
We show by induction on $j$ that, after processing $j$ jobs,
Algorithm~\ref{alg:fptas} stores a representative
$\hat s_j=(\hat C_i,\hat A_i,\hat B_i,\hat L_i)_{i=1}^K$ with
$\hat L_i=L_i^{j*}$ for all $i$ and
\[
\hat A_i\le A_i^{j*}\delta^{j},\qquad
\hat B_i\le B_i^{j*}\delta^{j},\qquad
\hat C_i\le C_i^{j*}\delta^{j}\qquad(1\le i\le K). \tag{$\star$}
\]
The base case $j=0$ is the stored zero state. Assume $(\star)$ holds for $j$,
and let $i^{*}$ be the block to which $s_j^{*}$ appends $J_{j+1}$. Because the
weight labels are stored exactly, block $i^{*}$ is empty in $\hat s_j$ whenever
it is empty in $s_j^{*}$, and otherwise
$w_{j+1}\le L_{i^{*}}^{j*}=\hat L_{i^{*}}$; hence the same transition is
applicable from $\hat s_j$. Applying it yields a candidate $\tilde s_{j+1}$
with $\tilde L_i=L_i^{j+1,*}$ for all $i$ and, by the recurrences of
Algorithm~\ref{alg:dp} and $\delta^{j}\ge 1$,
\begin{align*}
\tilde A_{i^{*}} &= \hat A_{i^{*}}+a_{j+1}
   \le \big(A_{i^{*}}^{j,*}+a_{j+1}\big)\delta^{j}=A_{i^{*}}^{j+1,*}\delta^{j},\\
\tilde B_{i^{*}} &= \hat B_{i^{*}}+b_{j+1}
   \le \big(B_{i^{*}}^{j,*}+b_{j+1}\big)\delta^{j}=B_{i^{*}}^{j+1,*}\delta^{j},\\
\tilde C_{i^{*}} &= \max\big\{\hat C_{i^{*}}+b_{j+1},\ \hat A_{i^{*}}+a_{j+1}+b_{j+1}\big\}\\
   &\le \max\big\{C_{i^{*}}^{j,*}+b_{j+1},\ A_{i^{*}}^{j,*}+a_{j+1}+b_{j+1}\big\}\delta^{j}
   = C_{i^{*}}^{j+1,*}\delta^{j},
\end{align*}
while all other coordinates are unchanged and satisfy the bound at $j+1$.
The candidate $\tilde s_{j+1}$ is mapped to some box $\mathscr{B}'$. By
interval dominance, the representative $\hat s_{j+1}$ ultimately stored in
$\mathscr{B}'$ satisfies $\hat x_i\le \delta\,\tilde x_i$ for every
$x\in\{A,B,C\}$ and every $i$, which yields $(\star)$ at $j+1$.

It remains to compare the objective values. In the block-concatenation
evaluation of Algorithm~\ref{alg:Block-concat}, the quantity $T_{B,i}$ is
nondecreasing in every coordinate and positively homogeneous, so
$T_{B,i}(\hat s_n)\le \delta^{n}\,T_{B,i}(s_n^{*})$. With
$\hat L_i=L_i^{n,*}$,
\[
W C_{\max}(\hat s_n)
=\max_{1\le i\le K}\hat L_i\,T_{B,i}(\hat s_n)
\le \delta^{n}\max_{1\le i\le K}L_i^{n,*}\,T_{B,i}(s_n^{*})
=\delta^{n}\,W C_{\max}^{*}.
\]
Finally, $\delta^{n}=(1+\varepsilon/(2n))^{n}\le 1+\varepsilon$ for
$0<\varepsilon<1$. The schedule returned by Algorithm~\ref{alg:fptas} is no
worse than the schedule realized by $\hat s_n$, so its value is at most
$(1+\varepsilon)\,W C_{\max}^{*}$.
\end{proof}

\begin{theorem}\label{thm:fptas-complex}
For fixed $K$, Algorithm~\ref{alg:fptas} runs in
\[
O\!\left(n^{3K+1}\,\frac{(\log V)^{3K}}{\varepsilon^{3K}}\right)
\]
time and is therefore a fully polynomial-time approximation scheme for
$F2 \parallel W C_{\max}$.
\end{theorem}

\begin{proof}
At each level $j$, Algorithm~\ref{alg:fptas} stores at most one representative
per box, and there are $(v+1)^{3K}(K+1)^{K}$ boxes. Each representative
generates $K$ candidates in $O(1)$ time, and the final evaluation of the
representatives at level $n$ costs $O(K)$ per box. Hence the running time is
\[
O\big(nK\,(v+1)^{3K}(K+1)^{K}\big).
\]
Since $\ln(1+\varepsilon/(2n))\ge\varepsilon/(4n)$ for $0<\varepsilon<1$,
\[
v=\left\lceil\frac{\ln U}{\ln\delta}\right\rceil
=O\!\left(\frac{n\log V}{\varepsilon}\right),
\]
and for constant $K$ the bound reduces to
$O(n^{3K+1}(\log V)^{3K}/\varepsilon^{3K})$, polynomial in $n$, $\log V$,
and $1/\varepsilon$.
\end{proof}
